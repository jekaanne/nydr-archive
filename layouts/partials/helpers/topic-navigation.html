{{- define "collect-sections" -}} {{- $page := .page -}} {{- $scratch := .scratch -}} {{- /* Only add image grouping / subtopic-gallery pages (skip docset-gallery landing pages) */ -}} {{- if eq $page.Params.layout "subtopic-gallery" -}} {{- $scratch.Add "list" (slice $page) -}} {{- end -}} {{- /* Recurse over child sections in weight order */ -}} {{- range $page.Sections.ByWeight -}} {{- template "collect-sections" (dict "page" . "scratch" $scratch) -}} {{- end -}} {{- /* Also include leaf bundle subtopic-gallery pages (e.g. index.md) in weight order */ -}} {{- range where $page.RegularPages.ByWeight "Params.layout" "subtopic-gallery" -}} {{- $scratch.Add "list" (slice .) -}} {{- end -}} {{- end -}} {{- $page := . -}} {{- if or (eq $page.Kind "section") (eq $page.Params.layout "subtopic-gallery") -}} {{- $scratch := newScratch -}} {{- $scratch.Set "list" slice -}} {{- /* Recursively collect subtopic-gallery pages in exhibition weight order */ -}} {{- template "collect-sections" (dict "page" (.Site.GetPage "section" "topics") "scratch" $scratch) -}} {{- $flatList := $scratch.Get "list" -}} {{- $currentIndex := -1 -}} {{- /* Find the current page index */ -}} {{- range $index, $element := $flatList -}} {{- if eq $element.RelPermalink $page.RelPermalink -}} {{- $currentIndex = $index -}} {{- end -}} {{- end -}} {{- $prevPage := "" -}} {{- $nextPage := "" -}} {{- if ge $currentIndex 0 -}} {{- $prevIndex := sub $currentIndex 1 -}} {{- $nextIndex := add $currentIndex 1 -}} {{- if ge $prevIndex 0 -}} {{- $prevPage = index $flatList $prevIndex -}} {{- end -}} {{- if lt $nextIndex (len $flatList) -}} {{- $nextPage = index $flatList $nextIndex -}} {{- end -}} {{- end -}}
{{- if $prevPage -}} Previous topic {{- end -}}
{{- if $nextPage -}} Next topic {{- end -}}
{{- end -}}
